3. C 对照：SetDevice→CreateContext→SetCurrentContext→malloc（真机 C 版本）
------------------------------------------------------------
cat > /tmp/ctx_alloc.c <<'EOF'
#include "acl/acl_rt.h"
#include <stdio.h>
#include <stdlib.h>
int main(void) {
    printf("ctx: start\n");
    int32_t r = aclInit(NULL);
    printf("ctx: aclInit ret=%d\n", (int)r);
    uint32_t cnt = 0;
    r = aclrtGetDeviceCount(&cnt);
    printf("ctx: cnt=%u ret=%d\n", cnt, (int)r);
    r = aclrtSetDevice(0);
    printf("ctx: setdev ret=%d\n", (int)r);
    aclrtContext ctx = NULL;
    r = aclrtCreateContext(&ctx, 0);
    printf("ctx: create ret=%d ctx=%p\n", (int)r, (void *)ctx);
    r = aclrtSetCurrentContext(ctx);
    printf("ctx: setcur ret=%d\n", (int)r);
    void *p = NULL;
    r = aclrtMalloc(&p, 1024, ACL_MEM_MALLOC_HUGE_FIRST);
    printf("ctx: malloc ret=%d ptr=%p\n", (int)r, (void *)p);
    if (r == ACL_SUCCESS) aclrtFree(p);
    printf("ctx: done\n");
    return 0;
}
EOF
gcc -o /tmp/ctx_alloc /tmp/ctx_alloc.c \
    -I/usr/local/Ascend/latest/include \
    -L/usr/local/Ascend/latest/lib64 -lascendcl \
    -Wl,-rpath,/usr/local/Ascend/latest/lib64
echo "编译退出码: $?"
