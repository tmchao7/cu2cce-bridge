3. 构造器探针：手动进 aclInit → SetDevice → malloc（不经过 wrapper）
但进程里仍带着 wrapper+aclblas → 看 malloc 是否 107002
------------------------------------------------------------
cat > /tmp/ctor_alloc.c <<'EOF'
#include "acl/acl_rt.h"
#include <stdio.h>
#include <stdlib.h>
int main(void) {
    printf("ctor_alloc: start\n");
    int32_t r = aclInit(nullptr);
    printf("ctor_alloc: aclInit ret=%d\n", (int)r);
    uint32_t cnt = 0;
    r = aclrtGetDeviceCount(&cnt);
    printf("ctor_alloc: cnt=%u ret=%d\n", cnt, (int)r);
    r = aclrtSetDevice(0);
    printf("ctor_alloc: setdev ret=%d\n", (int)r);
    void *p = nullptr;
    r = aclrtMalloc(&p, 1024, ACL_MEM_MALLOC_HUGE_FIRST);
    printf("ctor_alloc: malloc ret=%d ptr=%p\n", (int)r, (void *)p);
    if (r == ACL_SUCCESS) aclrtFree(p);
    return 0;
}
EOF
gcc -o /tmp/ctor_alloc /tmp/ctor_alloc.c \
    -I/usr/local/Ascend/latest/include \
    -L/usr/local/Ascend/latest/lib64 -lascendcl \
    -Wl,-rpath,/usr/local/Ascend/latest/lib64 \
    -L/home/t50063365/workspace/repos/cu2cce-bridge/build/real_test/install/lib \
    -lcce_bridge_runtime -Wl,-rpath,/home/t50063365/workspace/repos/cu2cce-bridge/build/real_test/install/lib
echo "ctor_alloc 编译退出码: $?"
/tmp/ctor_alloc
echo "ctor_alloc 运行退出码: $?"

【第 3 步 若 malloc ret=107002 → 坐实：wrapper（或其 aclblas 依赖）的
   构造器在 ACL 初始化前搞脏了全局 → 修复方向二选一：
     a) CMake 去掉 target_link_libraries 的 libacl_cblas（不链接 blas 库）
        b) 把 WCUDA_DEBUG 前的句子：把 aclblas 改为惰性 dlopen（用时才加载）
【第 3 步 若 malloc ret=0 → 构造器不是根因，wrapper 的运行时行为另有
   问题（比如 aclrtMalloc 参数在 wrapper 内被改）→ 下一步深挖 wrapper】
