# server_cmd.txt —— main / puresimt 官方 CUDA Samples 回归
#
# 维护分支：main；实时远端：gitcode（不要 push origin/GitHub）。
# 目标：用原始 NVIDIA CUDA Samples 源文件验证 puresimt 的零改写链路。
#
# 维护样例只有两个：
#   - examples/puresimt/vectorAdd/vectorAdd.cu
#   - examples/puresimt/matrixMul/matrixMul.cu
#
# 两个 .cu 文件保留 NVIDIA 原始源码，不添加 p 前缀、不加入 ACL 调用、不写
# Ascend C 语法。third_party/nvidia_cuda_samples/Common 只提供 NVIDIA 官方
# helper 头，作为 include 路径注入。
#
# 真机复制执行：
#   cd /home/t50063365/workspace/repos/cu2cce-bridge
#   git pull --ff-only gitcode main
#   bash tools/real_test/server_cmd.txt

set +e
set +u
set -o pipefail
ulimit -c 0 2>/dev/null || true

cd /home/t50063365/workspace/repos/cu2cce-bridge
printf 'HEAD=' && git pull --ff-only gitcode main && git log -1 --oneline

OUT="$PWD/build/real_test/puresimt"
mkdir -p "$OUT"
rm -f "$OUT"/*
FAIL=0
DEVICE_HEALTHY=0
DEVICE_ACCESS_DENIED=0
TEST_BLOCKED_BY_DEVICE=0

printf '\n===== [1/4] 只读设备健康门禁 =====\n'
for node in /dev/davinci0 /dev/davinci_manager /dev/hisi_hdc; do
    if [ ! -e "$node" ] || ! test -r "$node" || ! test -w "$node"; then
        echo "[BLOCKED] DEVICE_ACCESS_DENIED_OR_MISSING: $node"
        DEVICE_ACCESS_DENIED=1
    else
        echo "[PASS] device access: $node"
    fi
done

CANN_BASE="/usr/local/Ascend/latest"
if [ -f "$CANN_BASE/set_env.sh" ]; then
    # shellcheck source=/dev/null
    source "$CANN_BASE/set_env.sh"
elif [ -f "$CANN_BASE/bin/setenv.bash" ]; then
    # shellcheck source=/dev/null
    source "$CANN_BASE/bin/setenv.bash"
fi
export ASCEND_HOME_PATH="${ASCEND_HOME_PATH:-$CANN_BASE}"
export CANN_BASE

PROBE_DEVICE="$PWD/build/real_test/probe_device"
if [ ! -x "$PROBE_DEVICE" ]; then
    g++ tools/real_test/probe_device.cpp -o "$PROBE_DEVICE" \
        -I"$CANN_BASE/include" \
        -L"$CANN_BASE/lib64" \
        -Wl,-rpath,"$CANN_BASE/lib64" \
        -lascendcl -lpthread -ldl >"$OUT/probe_device_compile.log" 2>&1
fi
DEVICE_PROBE_FAIL=1
if [ -x "$PROBE_DEVICE" ]; then
    for attempt in 1 2; do
        if "$PROBE_DEVICE" >"$OUT/probe_device_${attempt}.log" 2>&1; then
            echo "[PASS] probe_device attempt=$attempt"
            DEVICE_PROBE_FAIL=0
            break
        else
            echo "[WARN] probe_device attempt=$attempt failed:"
            cat "$OUT/probe_device_${attempt}.log" 2>/dev/null
        fi
    done
else
    echo '[BLOCKED] probe_device executable missing; compile log:'
    cat "$OUT/probe_device_compile.log" 2>/dev/null
fi
if [ "$DEVICE_ACCESS_DENIED" -eq 0 ] && [ "$DEVICE_PROBE_FAIL" -eq 0 ]; then
    DEVICE_HEALTHY=1
    echo '[PASS] DEVICE_HEALTHY=1'
else
    TEST_BLOCKED_BY_DEVICE=1
    echo "[BLOCKED] TEST_BLOCKED_BY_DEVICE（ACCESS_DENIED=$DEVICE_ACCESS_DENIED PROBE_FAIL=$DEVICE_PROBE_FAIL）"
fi

printf '\n===== [2/4] CANN / bridge 环境 =====\n'
echo "ASCEND_HOME_PATH=$ASCEND_HOME_PATH"
echo "ASCEND_OPP_PATH=${ASCEND_OPP_PATH:-<unset>}"
REPO_INC="$PWD/include"
NVIDIA_COMMON="$PWD/third_party/nvidia_cuda_samples/Common"
REPO_LIB="$PWD/build/real_test/install/lib"
if [ ! -e "$REPO_LIB/libcce_bridge_runtime.so" ]; then
    REPO_LIB="$PWD/build/core/lib"
fi
export LD_LIBRARY_PATH="$REPO_LIB:$CANN_BASE/lib64:${LD_LIBRARY_PATH:-}"
echo "REPO_INC=$REPO_INC"
echo "NVIDIA_COMMON=$NVIDIA_COMMON"
echo "REPO_LIB=$REPO_LIB"
if [ ! -d "$NVIDIA_COMMON" ]; then
    echo '[FAIL] NVIDIA helper headers missing'
    FAIL=$((FAIL + 1))
fi

printf '\n===== [2.5/4] bridge runtime 重编 =====\n'
if bash tools/real_test/build_bridge.sh; then
    echo '[PASS] bridge runtime rebuild'
else
    echo '[FAIL] bridge runtime rebuild'
    FAIL=$((FAIL + 1))
fi
if [ -e "$PWD/build/real_test/install/lib/libcce_bridge_runtime.so" ]; then
    REPO_LIB="$PWD/build/real_test/install/lib"
fi
export LD_LIBRARY_PATH="$REPO_LIB:$CANN_BASE/lib64:${LD_LIBRARY_PATH:-}"

printf '\n===== [3/4] puresimt 官方 CUDA Samples 编译 + 运行 =====\n'
CCEC="$CANN_BASE/bin/ccec"
PS_FLAGS="-x asc --cce-aicore-arch=dav-c310-vec --enable-simt -std=c++17"
for pair in \
    'vectorAdd/vectorAdd.cu|Test PASSED' \
    'matrixMul/matrixMul.cu|Result = PASS'; do
    src=${pair%%|*}
    key=${pair#*|}
    name=$(basename "$src" .cu)
    bin="$OUT/$name"
    echo "===== [$name] $CCEC $PS_FLAGS examples/puresimt/$src ====="
    if "$CCEC" $PS_FLAGS "examples/puresimt/$src" \
        -I"$REPO_INC" -I"$NVIDIA_COMMON" -I"$CANN_BASE/include" \
        -L"$CANN_BASE/lib64" -L"$REPO_LIB" \
        -Wl,-rpath,"$REPO_LIB" -Wl,-rpath,"$CANN_BASE/lib64" \
        -lascendc_runtime -lascendcl -lruntime -lregister -lerror_manager \
        -lprofapi -lascendalog -lmmpa -lascend_dump -lc_sec -lstdc++ \
        -lcce_bridge_runtime -o "$bin" >"$OUT/${name}.log" 2>&1; then
        ldd "$bin" >"$OUT/${name}.ldd.log" 2>&1 || true
        if [ "$DEVICE_HEALTHY" -ne 1 ]; then
            echo "[SKIP] [$name] 编译通过；设备门禁未通过，未执行二进制"
        elif "$bin" >"$OUT/${name}.run.log" 2>&1 && grep -Fq "$key" "$OUT/${name}.run.log"; then
            echo "[PASS] [$name] 运行通过：$(grep -F "$key" "$OUT/${name}.run.log" | tail -n 1)"
        else
            echo "[FAIL] [$name] 运行失败（run log 首 30 行）"
            head -n 30 "$OUT/${name}.run.log" 2>/dev/null
            tail -n 20 "$OUT/${name}.log" 2>/dev/null
            FAIL=$((FAIL + 1))
        fi
    else
        echo "[FAIL] [$name] 编译失败（log 首 30 行）"
        head -n 30 "$OUT/${name}.log" 2>/dev/null
        FAIL=$((FAIL + 1))
    fi
done

printf '\n===== [4/4] 汇总 =====\n'
echo "FAIL=$FAIL DEVICE_HEALTHY=$DEVICE_HEALTHY TEST_BLOCKED_BY_DEVICE=$TEST_BLOCKED_BY_DEVICE"
echo "完整日志：$OUT"
if [ "$TEST_BLOCKED_BY_DEVICE" -eq 1 ]; then
    echo 'RESULT=TEST_BLOCKED_BY_DEVICE'
elif [ "$FAIL" -eq 0 ]; then
    echo 'PURESIMT_REGRESSION_PASS'
else
    echo 'PURESIMT_REGRESSION_FAIL（查看上方各 .log）'
fi
