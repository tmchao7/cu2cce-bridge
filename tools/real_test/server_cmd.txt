# server_cmd.txt —— B 方案真机探针：隐藏 bx/by/bz 动态传给 async_invoke
#
# 分支：exp/hidden-blockdim-args
# 目的：只验证普通 wrapper 参数能否用于 cce::dim3{bx,by,bz}。
# 若通过，同一个 wrapper 将可在运行时分别使用 256、512 和 2D block；
# 若失败，则 B 方案淘汰，转测 A 静态推导分支。
#
# 服务器先切到本分支：
#   cd /home/t50063365/workspace/repos/cu2cce-bridge
#   git fetch origin
#   git switch -C exp/hidden-blockdim-args --track origin/exp/hidden-blockdim-args
#   cat tools/real_test/server_cmd.txt
# 然后将下方命令块整段复制执行，并贴回完整输出。

set -eo pipefail
ulimit -c 0 2>/dev/null || true

cd /home/t50063365/workspace/repos/cu2cce-bridge
git pull --ff-only origin exp/hidden-blockdim-args
printf 'HEAD=' && git log -1 --oneline

OUT="$PWD/build/real_test/hidden_blockdim_args"
mkdir -p "$OUT"
rm -f "$OUT"/*

printf '\n===== [1/5] 本地结构测试 =====\n'
python3 -m unittest discover -s tests -p 'test_*.py'

printf '\n===== [2/5] 环境探测与安装 =====\n'
bash tools/real_test/env_probe.sh
bash tools/real_test/build_bridge.sh
export PYTHONPATH="${PYTHONPATH:-}"
export CMAKE_PREFIX_PATH="${CMAKE_PREFIX_PATH:-}"
source "$PWD/build/real_test/probe.env"
export CUCC_PATH="$PWD/build/real_test/install"
export CUDA_PATH="$CUCC_PATH"
export PATH="$CUCC_PATH/bin:$PATH"
export LD_LIBRARY_PATH="$CUCC_PATH/lib:$ASCEND_LIB:${LD_LIBRARY_PATH:-}"
if [ -n "${SIMT_API_INCLUDE:-}" ]; then
    export CPLUS_INCLUDE_PATH="${SIMT_API_ROOT:-$(dirname "$SIMT_API_INCLUDE")}:$SIMT_API_INCLUDE:${CPLUS_INCLUDE_PATH:-}"
fi

printf '\n===== [3/5] 检查转换结果 =====\n'
python3 tools/kernel_convert.py \
    examples/experimental/hiddenBlockDim/hidden_blockdim_probe.cu \
    -o "$OUT/hidden_blockdim_probe.cce"
grep -nF 'cce::dim3{bx, by, bz}' "$OUT/hidden_blockdim_probe.cce"
grep -nF 'hiddenAdd<<<1, threads256, stream>>>' "$OUT/hidden_blockdim_probe.cce"
grep -nF 'hiddenAdd<<<1, threads512, stream>>>' "$OUT/hidden_blockdim_probe.cce"
grep -nF 'hidden2D<<<1, block2d, stream>>>' "$OUT/hidden_blockdim_probe.cce"
if grep -Fq 'cce::dim3{blockDim.x, blockDim.y, blockDim.z}' "$OUT/hidden_blockdim_probe.cce"; then
    echo 'FAIL：wrapper 仍读取 blockDim' >&2
    exit 1
fi

printf '\n===== [4/5] 编译、链接、运行 B 探针 =====\n'
if WCUDA_DEBUG=1 "$CUCC_PATH/bin/cucc" -c \
       examples/experimental/hiddenBlockDim/hidden_blockdim_probe.cu \
       -o "$OUT/hidden_blockdim_probe.o" >"$OUT/compile.log" 2>&1; then
    echo '[PASS] B 探针编译通过'
else
    rc=$?
    echo "[FAIL] B 探针编译失败 rc=$rc"
    grep -nEm 12 'error:|fatal error:|Segmentation fault|stack smashing|Assertion|Stack dump' "$OUT/compile.log" || true
    tail -n 40 "$OUT/compile.log"
    exit "$rc"
fi

"$CUCC_PATH/bin/cucc" "$OUT/hidden_blockdim_probe.o" \
    -o "$OUT/hidden_blockdim_probe" >"$OUT/link.log" 2>&1
"$OUT/hidden_blockdim_probe" | tee "$OUT/run.log"
grep -F 'hidden blockdim OK (256,512,2d)' "$OUT/run.log"

printf '\n===== [5/5] 判读 =====\n'
echo '若看到 hidden blockdim OK (256,512,2d)：B 方案通过，可将 block 表达式作为隐藏参数传给单一 wrapper。'
echo '若 cce::dim3{bx,by,bz} 编译崩溃或报必须为常量：B 方案失败，下一步测试 A 静态推导。'
echo "完整日志：$OUT"
