compile_case() {
    name="$1"
    src="$2"
    if "$CUCC_PATH/bin/cucc" -c "$src" -o "$OUT/$name.o" >"$OUT/$name.compile.log" 2>&1; then
        echo "[PASS] $name compile"
    else
        rc=$?
        echo "[FAIL] $name compile rc=$rc"
        grep -nEm 12 'error:|fatal error:|Segmentation fault|stack smashing|Assertion|Stack dump' "$OUT/$name.compile.log" || true
        tail -n 35 "$OUT/$name.compile.log"
        FAIL=$((FAIL + 1))
    fi
}
compile_case vectorAdd examples/vectorAdd/vectorAdd.cu
compile_case minimal examples/minimal/minimal.cu
compile_case dynamic_blockdim examples/experimental/dynamicBlockDim/dynamic_blockdim.cu

printf '\n===== [5/7] 正式样例链接与运行 =====\n'
run_case() {
    name="$1"
    expected="$2"
    if [ ! -f "$OUT/$name.o" ]; then
        echo "[SKIP] $name object missing"
        FAIL=$((FAIL + 1))
        return
    fi
    if "$CUCC_PATH/bin/cucc" "$OUT/$name.o" -o "$OUT/$name" >"$OUT/$name.link.log" 2>&1 \
       && "$OUT/$name" >"$OUT/$name.run.log" 2>&1 \
       && grep -Fq "$expected" "$OUT/$name.run.log"; then
        echo "[PASS] $name: $(cat "$OUT/$name.run.log")"
    else
        echo "[FAIL] $name link/run"
        tail -n 30 "$OUT/$name.link.log" 2>/dev/null || true
        tail -n 30 "$OUT/$name.run.log" 2>/dev/null || true
        FAIL=$((FAIL + 1))
    fi
}
run_case vectorAdd 'vectorAdd OK (N=1024)'
run_case minimal 'minimal OK (N=256)'
run_case dynamic_blockdim 'dynamic blockdim OK (256,512,32x8x1)'
