    bin="$OUT/$name"

    INCLUDE_FLAGS=( -I"$REPO_INC" -I"$CANN_BASE/include" )

    # matrixMul includes the system C++ math header itself, so do not preinclude

        vectorAdd)
            EXTRA_FLAGS+=( -include "$REPO_INC/shadow/math.h" )
            INCLUDE_FLAGS+=( -I"$NVIDIA_COMMON" )
            ;;
        matrixMul)
            INCLUDE_FLAGS+=( -I"$NVIDIA_COMMON" )
            ;;
        math_minimal)
            EXTRA_FLAGS+=( -include "$REPO_INC/shadow/math.h" )
            ;;
    esac
    if "$CCEC" $PS_FLAGS "${EXTRA_FLAGS[@]}" "examples/puresimt/$src" \
        "${INCLUDE_FLAGS[@]}" \
        -L"$CANN_BASE/lib64" -L"$REPO_LIB" \
        -Wl,-rpath,"$REPO_LIB" -Wl,-rpath,"$CANN_BASE/lib64" \

        -lprofapi -lascendalog -lmmpa -lascend_dump -lc_sec -lstdc++ \
        -lcce_bridge_runtime -o "$bin" >"$OUT/${name}.log" 2>&1; then

            echo "[SKIP] [$name] 编译通过；设备门禁未通过，未执行二进制"
        elif "$bin" >"$OUT/${name}.run.log" 2>&1 && grep -Fq "$key" "$OUT/${name}.run.log"; then
            echo "[PASS] [$name] 运行通过：$(grep -F "$key" "$OUT/${name}.run.log" | tail -n 1)"
        else
            echo "[FAIL] [$name] 运行失败（run log 首 30 行）"
            head -n 30 "$OUT/${name}.run.log" 2>/dev/null
            echo "----- [$name] ldd（仅动态库依赖，不是运行错误） -----"
            cat "$OUT/${name}.ldd.log" 2>/dev/null
            tail -n 20 "$OUT/${name}.log" 2>/dev/null
            FAIL=$((FAIL + 1))
        fi
    else
        echo "[FAIL] [$name] 编译失败（log 首 30 行）"
        head -n 30 "$OUT/${name}.log" 2>/dev/null
        FAIL=$((FAIL + 1))
    fi
done

run_cmake_case() {
    local name="$1"
    local source_dir="$2"
    local build_dir="$3"
    local binary="$4"
    local expected="$5"
    local extra_args=("${@:6}")
    local log_prefix="$OUT/$name"
    local configure_log="${log_prefix}_configure.log"
    local build_log="${log_prefix}_build.log"
    local run_log="${log_prefix}_run.log"

    rm -rf "$build_dir"
    echo "----- [$name] configure: $source_dir -----"
    if "$CMAKE_CCE" -S "$source_dir" -B "$build_dir" "${extra_args[@]}" \
        -DCMAKE_CUDA_ARCHITECTURES=OFF >"$configure_log" 2>&1; then
        echo "[PASS] [$name] CMake configure"
    else
        echo "[FAIL] [$name] CMake configure (log: $configure_log)"
        tail -n 100 "$configure_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    if cmake --build "$build_dir" >"$build_log" 2>&1; then
        echo "[PASS] [$name] CMake build"
    else
        echo "[FAIL] [$name] CMake build (log: $build_log)"
        tail -n 100 "$build_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    if "$binary" >"$run_log" 2>&1 && grep -Fq "$expected" "$run_log"; then
        echo "[PASS] [$name] CMake run: $(grep -F "$expected" "$run_log" | tail -n 1)"
    else
        echo "[FAIL] [$name] CMake run (log: $run_log)"
        cat "$run_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    if grep -Eiq 'kernel[_ -]?convert|__simt_vf__|async_invoke|\.cce' \
        "$configure_log" "$build_log"; then
        echo "[FAIL] [$name] CMake path emitted legacy lowering markers"
        CMAKE_PASS=0
    else
        echo "[PASS] [$name] no legacy lowering markers"
    fi
}
