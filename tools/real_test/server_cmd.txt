python3 - "$DIAG_DIR" <<'PY'
from pathlib import Path
import sys

d = Path(sys.argv[1])
dynamic = "cce::dim3{blockDim.x, blockDim.y, blockDim.z}"

vector = (d / "vector_dynamic512.cu").read_text()
minimal = (d / "minimal_dynamic512.cu").read_text()
assert dynamic in vector and dynamic in minimal
assert "int threads = 512;" in vector
assert "addOne<<<1, 512, s>>>" in minimal

# vector：动态/静态 wrapper × 256/512 host launch，严格一次只改一个变量。
(d / "vector_static512.cu").write_text(
    vector.replace(dynamic, "cce::dim3{512, 1, 1}")
)
(d / "vector_dynamic256.cu").write_text(
    vector.replace("int threads = 512;", "int threads = 256;")
)
(d / "vector_static256.cu").write_text(
    vector.replace(dynamic, "cce::dim3{256, 1, 1}")
          .replace("int threads = 512;", "int threads = 256;")
)

# 第二个独立样例：minimal 动态 wrapper 与静态 512 wrapper 对照。
(d / "minimal_static512.cu").write_text(
    minimal.replace(dynamic, "cce::dim3{512, 1, 1}")
)
PY

for f in "$DIAG_DIR"/*.cu; do

printf '\n===== [4/6] 只编译六个对照（失败后继续收集） =====\n'
PASS=0
FAIL=0
run_compile_case() {
    local name="$1"
    local src="$DIAG_DIR/${name}.cu"
    local obj="$DIAG_DIR/${name}.o"
    local log="$DIAG_DIR/${name}.compile.log"
    echo "----- $name -----"
    if WCUDA_DEBUG=1 "$CUCC_PATH/bin/cucc" -c "$src" -o "$obj" >"$log" 2>&1; then
        echo "  [PASS] $name"
        PASS=$((PASS + 1))
    else
        local rc=$?
        echo "  [FAIL] $name rc=$rc log=$log"
        echo "  --- 首个诊断/崩溃标记 ---"
        grep -nEm 8 'error:|fatal error:|Segmentation fault|stack smashing|Assertion|Stack dump' "$log" || true
        echo "  --- 日志末尾 25 行 ---"
        tail -n 25 "$log" | sed 's/^/      /'
        FAIL=$((FAIL + 1))
    fi
}

run_compile_case vector_dynamic512
run_compile_case vector_static512
run_compile_case vector_dynamic256
run_compile_case vector_static256
run_compile_case minimal_dynamic512
run_compile_case minimal_static512

echo "COMPILE_SUMMARY PASS=$PASS FAIL=$FAIL"

printf '\n===== [5/6] 对所有编译成功的两个 512 静态样例做链接+数值运行 =====\n'
run_link_and_execute() {
    local name="$1"
    local expected="$2"
    local obj="$DIAG_DIR/${name}.o"
    local bin="$DIAG_DIR/${name}"
    local log="$DIAG_DIR/${name}.run.log"
    if [ ! -f "$obj" ]; then
        echo "  [SKIP] $name：编译未通过"
        return 0
    fi
    echo "----- ${name}_link_run -----"
    if "$CUCC_PATH/bin/cucc" "$obj" -o "$bin" >"$DIAG_DIR/${name}.link.log" 2>&1 \
       && "$bin" >"$log" 2>&1 \
       && grep -Fq "$expected" "$log"; then
        echo "  [PASS] $name：$(cat "$log")"
    else
        echo "  [FAIL] $name 链接或数值运行失败"
        tail -n 30 "$DIAG_DIR/${name}.link.log" 2>/dev/null || true
        tail -n 30 "$log" 2>/dev/null || true
    fi
}

run_link_and_execute vector_static512 'vectorAdd OK (N=1024)'
run_link_and_execute minimal_static512 'minimal OK (N=256)'
