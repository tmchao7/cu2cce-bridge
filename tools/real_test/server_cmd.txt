resolve_regression_remote() {
    local requested="${CU2CCE_REGRESSION_REMOTE:-}"
    if [ -n "$requested" ]; then
        if git remote get-url "$requested" >/dev/null 2>&1; then
            printf '%s\n' "$requested"
            return 0
        fi
        echo "[FAIL] requested regression remote '$requested' does not exist" >&2
        return 1
    fi
    if git remote get-url gitcode >/dev/null 2>&1; then
        printf '%s\n' gitcode
        return 0
    fi
    if git remote get-url origin >/dev/null 2>&1; then
        printf '%s\n' origin
        return 0
    fi
    echo '[FAIL] no usable git remote named gitcode or origin' >&2
    return 1
}

artifact_check() {
    local case_name="$1"
    local root="$2"
    local log="$ARTIFACT_LOG_DIR/${case_name}.artifact.log"
    local output="$ARTIFACT_LOG_DIR/${case_name}.artifact.stdout"
    local error="$ARTIFACT_LOG_DIR/${case_name}.artifact.stderr"
    ARTIFACT_CASES_EXECUTED=1
    python3 tools/real_test/verify_artifacts.py \
        --contract "$ARTIFACT_CONTRACT" \
        --case "$case_name" \
        --root "$root" \
        --log "$log" >"$output" 2>"$error"
    local status=$?
    if [ "$status" -eq 0 ]; then
        cat "$output"
        return 0
    fi
    ARTIFACT_CONTRACT_PASS=0
    echo "[FAIL] artifact [$case_name] (details: $log)"
    tail -n 30 "$log" 2>/dev/null || true
    return "$status"
}

run_cmake_case() {
    local name="$1"
    local source_dir="$2"
    local build_dir="$3"
    local binary="$4"
    local expected="$5"
    local extra_args=("${@:6}")
    local log_prefix="$OUT/$name"
    local configure_log="${log_prefix}_configure.log"
    local build_log="${log_prefix}_build.log"
    local run_log="${log_prefix}_run.log"

    rm -rf "$build_dir"
    echo "----- [$name] configure: $source_dir -----"
    if "$CMAKE_CCE" -S "$source_dir" -B "$build_dir" "${extra_args[@]}" \
        -DCMAKE_CUDA_ARCHITECTURES=OFF >"$configure_log" 2>&1; then
        echo "[PASS] [$name] CMake configure"
    else
        echo "[FAIL] [$name] CMake configure (log: $configure_log)"
        tail -n 100 "$configure_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    # Configure uses mock_nvcc; build must go through cmake_cce so it
    # repoints the persistent virtual toolkit and executes the real cucc path.
    if "$CMAKE_CCE" --build "$build_dir" >"$build_log" 2>&1; then
        echo "[PASS] [$name] CMake build"
    else
        echo "[FAIL] [$name] CMake build (log: $build_log)"
        tail -n 100 "$build_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    # The configure virtual toolkit entry is intentionally mock-backed. Verify this build
    # actually crossed the second phase and repointed nvcc to real cucc before
    # accepting its artifacts; otherwise a mock-only build could look green.
    if ! grep -Fq 'cmake_cce: build phase: virtual toolkit nvcc ->' "$build_log" || \
       [ ! -L "$CU2CCE_CMAKE_SHIM_DIR/CUDA_DIR/bin/nvcc" ] || \
       [ "$(readlink "$CU2CCE_CMAKE_SHIM_DIR/CUDA_DIR/bin/nvcc")" != "$CUCC_PATH/bin/cucc" ]; then
        echo "[FAIL] [$name] CMake build did not use real cucc (log: $build_log)"
        tail -n 30 "$build_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    if ! artifact_check "$name" "$build_dir"; then
        echo "[FAIL] [$name] CMake artifact contract"
        CMAKE_PASS=0
        return 0
    fi

    if "$binary" >"$run_log" 2>&1 && grep -Fq "$expected" "$run_log"; then
        echo "[PASS] [$name] CMake run: $(grep -F "$expected" "$run_log" | tail -n 1)"
    else
        echo "[FAIL] [$name] CMake run (log: $run_log)"
        tail -n 30 "$run_log" 2>/dev/null || true
        CMAKE_PASS=0
        return 0
    fi

    if grep -Eiq 'kernel[_ -]?convert|__simt_vf__|async_invoke|\.cce' \
        "$configure_log" "$build_log"; then
        echo "[FAIL] [$name] CMake path emitted legacy lowering markers"
        CMAKE_PASS=0
    else
        echo "[PASS] [$name] no legacy lowering markers"
    fi
}

    # -lcudart, proving the CUDA-named lib64 symlink family (plain-g++
    # Make projects are a common large-project shape).
    rm -rf examples/build_compat/make_gxx_link/build
    if CUCC_PATH="$PWD/build/real_test/install" CUBRIDGE_HOME="$OUT/make-home" \
        "$MAKE_CCE" -C examples/build_compat/make_gxx_link >"$OUT/make_gxx_link.log" 2>&1; then
        if examples/build_compat/make_gxx_link/build/app >"$OUT/make_gxx_link.run.log" 2>&1 && \
           grep -Fq 'make_gxx_link OK' "$OUT/make_gxx_link.run.log"; then
            echo '[PASS] make_gxx_link: plain-g++ link with -lcudart -> bridge runtime'
            if ! artifact_check make_gxx_link "$PWD/examples/build_compat/make_gxx_link"; then
                BUILD_COMPAT_PASS=0
            fi
        else
            echo '[FAIL] make_gxx_link run'
            tail -n 30 "$OUT/make_gxx_link.run.log" 2>/dev/null || true
            BUILD_COMPAT_PASS=0
        fi
    else
        echo '[FAIL] make_gxx_link build (log tail:)'
        tail -n 60 "$OUT/make_gxx_link.log" 2>/dev/null || true
        BUILD_COMPAT_PASS=0
    fi
else
    echo '[FAIL] make_cce smoke unavailable'
    BUILD_COMPAT_PASS=0
fi
if [ "${BUILD_COMPAT_PASS:-0}" -eq 1 ] && [ "${CMAKE_PASS:-0}" -eq 1 ] && command -v ninja >/dev/null 2>&1 && [ -x "$NINJA_CCE" ]; then
    if CUCC_PATH="$PWD/build/real_test/install" "$NINJA_CCE" -C "$OUT/cmake-ninja-build" >"$OUT/ninja_cce.log" 2>&1 &&
       grep -Fq 'ninja_cce: build phase: virtual toolkit nvcc ->' "$OUT/ninja_cce.log"; then
        echo '[PASS] direct ninja_cce wrapper + persistent virtual toolkit rebuild'
        if ! artifact_check cmake_ninja "$OUT/cmake-ninja-build"; then
            BUILD_COMPAT_PASS=0
        fi
        if [ -d "$OUT/cmake-build-compat-ninja-build" ]; then
            if CUCC_PATH="$PWD/build/real_test/install" "$NINJA_CCE" -C "$OUT/cmake-build-compat-ninja-build" \
                >"$OUT/ninja_cce_build_compat.log" 2>&1 &&
               grep -Fq 'ninja_cce: build phase: virtual toolkit nvcc ->' "$OUT/ninja_cce_build_compat.log" &&
               artifact_check cmake_build_compat_ninja "$OUT/cmake-build-compat-ninja-build"; then
                echo '[PASS] direct ninja_cce build-compat artifact shape matches CMake Makefiles'
            else
                echo '[FAIL] direct ninja_cce build-compat artifact contract'
                tail -n 60 "$OUT/ninja_cce_build_compat.log" 2>/dev/null || true
                BUILD_COMPAT_PASS=0
            fi
        fi
    else
        echo '[FAIL] ninja_cce wrapper'
        tail -n 100 "$OUT/ninja_cce.log" 2>/dev/null
        BUILD_COMPAT_PASS=0
    fi
fi
fi
mark_timing make_ninja_suite

elif [ "${BUILD_COMPAT_PASS:-0}" -eq 1 ]; then

printf '\n===== [7/7] External NVIDIA CUDA Samples CMake smoke =====\n'
EXTERNAL_FETCH="$PWD/tools/real_test/fetch_external_cuda_samples.sh"
# Source cache lives OUTSIDE build/ on purpose: the standard pull flow runs
# clean.sh, which wipes build/, and the fetch script reuses a tree whose
# .cu2cce_revision matches the pin — so a persistent cache turns every
# post-clean run into a zero-network hit.
EXTERNAL_SOURCE="${CU2CCE_EXTERNAL_CACHE_DIR:-$PWD/.cu2cce-cache}/nvidia_cuda_samples"
EXTERNAL_BUILD="$OUT/cuda-samples-build"
EXTERNAL_LOG="$OUT/cuda_samples.log"
EXTERNAL_BUILD_LOG="$OUT/cuda_samples_build.log"
EXTERNAL_FETCH_TIMEOUT="${CU2CCE_EXTERNAL_FETCH_TIMEOUT:-300}"
EXTERNAL_BUILD_TIMEOUT="${CU2CCE_EXTERNAL_BUILD_TIMEOUT:-1200}"
EXTERNAL_RUN_TIMEOUT="${CU2CCE_EXTERNAL_RUN_TIMEOUT:-180}"
EXTERNAL_JOBS="$(nproc 2>/dev/null || echo 4)"

run_with_timeout() {
    local seconds="$1"
    shift
    if command -v timeout >/dev/null 2>&1; then
        timeout --foreground "$seconds" "$@"
    else
        echo "[WARN] timeout command unavailable; running without time limit: $*" >&2
        "$@"
    fi
}

# Incremental build tree: keep it across runs and let cmake decide what to
# rebuild. CMake validates all paths on configure; a stale tree that cannot
# be salvaged falls back to a clean reconfigure automatically.
mkdir -p "$EXTERNAL_BUILD"
CUDA_SAMPLES_EXECUTED=0
if ! profile_runs "external,full"; then
    echo "[SKIP] external NVIDIA CUDA Samples section (profile=$REGRESSION_PROFILE)"
elif [ "$DEVICE_HEALTHY" -eq 1 ] && [ -x "$EXTERNAL_FETCH" ] && [ -x "$CMAKE_CCE" ]; then
    CUDA_SAMPLES_EXECUTED=1
    CUDA_SAMPLES_PASS=1
    echo "[INFO] external CUDA Samples fetch (timeout=${EXTERNAL_FETCH_TIMEOUT}s)"
    if run_with_timeout "$EXTERNAL_FETCH_TIMEOUT" env \
       OUT_DIR="$OUT" CUDA_SAMPLES_SOURCE_DIR="$EXTERNAL_SOURCE" \
       "$EXTERNAL_FETCH" >"$EXTERNAL_LOG" 2>&1; then
        echo '[PASS] external CUDA Samples fetch'
    else
        echo '[FAIL] external CUDA Samples fetch (log: '$EXTERNAL_LOG')'
        tail -n 120 "$EXTERNAL_LOG" 2>/dev/null
        CUDA_SAMPLES_PASS=0
    fi

    if [ "$CUDA_SAMPLES_PASS" -eq 1 ]; then
        echo "[INFO] external CUDA Samples configure/build (build timeout=${EXTERNAL_BUILD_TIMEOUT}s, jobs=${EXTERNAL_JOBS})"
        if env CUCC_PATH="$PWD/build/real_test/install" CUDA_PATH="$CUCC_PATH" \
           PATH="$CUCC_PATH/bin:$PATH" CU2CCE_LIBC_MODE=system \
           "$CMAKE_CCE" -S examples/external/cuda_samples_project -B "$EXTERNAL_BUILD" \
             -DCUDA_SAMPLES_SOURCE_DIR="$EXTERNAL_SOURCE" \
             -DCMAKE_CUDA_ARCHITECTURES=OFF -DCMAKE_BUILD_TYPE=Release \
             >"$OUT/cuda_samples_configure.log" 2>&1; then
            echo '[PASS] external CUDA Samples configure'
            if run_with_timeout "$EXTERNAL_BUILD_TIMEOUT" env \
               CU2CCE_LIBC_MODE=system CUCC_PATH="$PWD/build/real_test/install" CUDA_PATH="$CUCC_PATH" \
               PATH="$CUCC_PATH/bin:$PATH" \
               "$CMAKE_CCE" --build "$EXTERNAL_BUILD" --parallel "$EXTERNAL_JOBS" --target vectorAdd matrixMul \
                >"$EXTERNAL_BUILD_LOG" 2>&1; then
                echo '[PASS] external CUDA Samples build'
                for sample in vectorAdd matrixMul; do
                    binary="$EXTERNAL_BUILD/$sample/$sample"
                    run_log="$OUT/cuda_samples_${sample}.run.log"
                    if [ "$sample" = vectorAdd ]; then expect='Test PASSED'; fi
                    if [ "$sample" = matrixMul ]; then expect='Result = PASS'; fi
                    echo "[INFO] external [$sample] run (timeout=${EXTERNAL_RUN_TIMEOUT}s)"
                    if [ -x "$binary" ] && run_with_timeout "$EXTERNAL_RUN_TIMEOUT" "$binary" >"$run_log" 2>&1 && grep -Fq "$expect" "$run_log"; then
                        echo "[PASS] external [$sample]: $(grep -F "$expect" "$run_log" | tail -n 1) (details: $run_log)"
                    else
                        echo "[FAIL] external [$sample] run (log: $run_log)"
                        tail -n 30 "$run_log" 2>/dev/null || true
                        CUDA_SAMPLES_PASS=0
                    fi
                done
                # Diagnostic-only samples stay in the external configure/build
                # graph but never block the required profile. They are NOT part
                # of the required "--target vectorAdd matrixMul" invocation, so
                # each one is compiled separately here before running; an
                # unsupported sample surfaces its true failure point (compile vs
                # run) through the per-sample logs instead of masking as a
                # silent no-op.
                printf '\n----- external diagnostic-only samples -----\n'
                for sample in simpleStreams simpleMultiCopy asyncAPI; do
                    binary="$EXTERNAL_BUILD/$sample/$sample"
                    diag_build_log="$OUT/cuda_samples_${sample}.build.log"
                    diag_log="$OUT/cuda_samples_diag_${sample}.log"
                    echo "[INFO] external diagnostic [$sample] build (timeout=${EXTERNAL_BUILD_TIMEOUT}s)"
                    if [ -x "$binary" ]; then
                        echo "[INFO] external diagnostic [$sample] binary already present"
                    elif ! run_with_timeout "$EXTERNAL_BUILD_TIMEOUT" env \
                       CU2CCE_LIBC_MODE=system CUCC_PATH="$PWD/build/real_test/install" CUDA_PATH="$CUCC_PATH" \
                       PATH="$CUCC_PATH/bin:$PATH" \
                       "$CMAKE_CCE" --build "$EXTERNAL_BUILD" --parallel "$EXTERNAL_JOBS" --target "$sample" >"$diag_build_log" 2>&1; then
                        echo "[DIAG-ONLY] [$sample] build failed (log: $diag_build_log; tail 12)"
                        tail -n 12 "$diag_build_log" 2>/dev/null || true
                        continue
                    fi
                    if [ ! -x "$binary" ]; then
                        echo "[DIAG-ONLY] [$sample] binary missing after build step"
                        continue
                    fi
                    echo "[INFO] external diagnostic [$sample] run (timeout=${EXTERNAL_RUN_TIMEOUT}s)"
                    # Capture the real exit status: an empty log with a nonzero
                    # status is itself evidence (crash/abort vs. timeout=124).
                    # WCUDA_DEBUG is forced ON for diagnostic-only runs so a
                    # failing sample leaves a per-call bridge trace
                    # (tools/real_test/diag/（原工作区历史材料未随仓库交付）). Export
                    # form, not env prefix: run_with_timeout is a shell
                    # function and env cannot launch functions.
                    WCUDA_DEBUG=1 run_with_timeout "$EXTERNAL_RUN_TIMEOUT" "$binary" >"$diag_log" 2>&1
                    diag_status=$?
                    if [ "$diag_status" -eq 0 ]; then
                        echo "[DIAG-PASS] [$sample] completed (unexpectedly passed; re-classify or investigate API surface; details: $diag_log)"
                    else
                        echo "[DIAG-ONLY] [$sample] failed, exit=$diag_status (log: $diag_log; tail 12)"
                        tail -n 12 "$diag_log" 2>/dev/null || true
                    fi
                done
                if grep -Eiq 'kernel[_ -]?convert|__simt_vf__|async_invoke|\.cce' \
                    "$OUT/cuda_samples_configure.log" "$EXTERNAL_BUILD_LOG"; then
                    echo '[FAIL] external CUDA Samples emitted legacy lowering markers'
                    CUDA_SAMPLES_PASS=0
                fi
            else
                echo "[FAIL] external CUDA Samples build (log: $EXTERNAL_BUILD_LOG)"
                tail -n 160 "$EXTERNAL_BUILD_LOG" 2>/dev/null
                CUDA_SAMPLES_PASS=0
            fi
        else
            echo '[FAIL] external CUDA Samples configure'
            tail -n 160 "$OUT/cuda_samples_configure.log" 2>/dev/null
            CUDA_SAMPLES_PASS=0
        fi
    fi
else
    echo '[FAIL] external CUDA Samples smoke unavailable'
    CUDA_SAMPLES_PASS=0
fi
mark_timing external_samples

if [ "$ARTIFACT_CASES_EXECUTED" -eq 1 ]; then
