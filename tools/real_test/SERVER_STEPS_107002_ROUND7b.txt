#include "acl/acl_rt.h"
#include <stdio.h>
#include <stdlib.h>
int main(void) {
    printf("ctor_alloc: start\n");
    int32_t r = aclInit(NULL);
    printf("ctor_alloc: aclInit ret=%d\n", (int)r);
    uint32_t cnt = 0;
    r = aclrtGetDeviceCount(&cnt);
    printf("ctor_alloc: cnt=%u ret=%d\n", cnt, (int)r);
    r = aclrtSetDevice(0);
    printf("ctor_alloc: setdev ret=%d\n", (int)r);
    void *p = NULL;
    r = aclrtMalloc(&p, 1024, ACL_MEM_MALLOC_HUGE_FIRST);
    printf("ctor_alloc: malloc ret=%d ptr=%p\n", (int)r, (void *)p);
    if (r == ACL_SUCCESS) aclrtFree(p);
    return 0;
}
EOF
