cmake_minimum_required(VERSION 3.18)
project(cu2cce_build_compat LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_library(build_compat_kernels STATIC kernel_a.cu kernel_b.cu)
add_executable(build_compat_app main.cpp)
target_link_libraries(build_compat_app PRIVATE build_compat_kernels)
install(TARGETS build_compat_app EXPORT buildCompatTargets)
install(EXPORT buildCompatTargets DESTINATION lib/cmake/build_compat)
