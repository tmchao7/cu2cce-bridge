# ---------- 3. 编译你自己的 CUDA 程序 ----------
# 设置环境（每次新终端先执行）
export CUCC_PATH=$PWD/build/real_test/install
export CUDA_PATH=$CUCC_PATH
export PATH=$CUCC_PATH/bin:$PATH

# 方式 A：两步法（推荐）
cucc -c my_program.cu -o my_program.o
cucc my_program.o -o my_program

# 方式 B：一步法
cucc my_program.cu -o my_program

# 运行（device 0；rpath 已内置，无需 LD_LIBRARY_PATH）
./my_program

# 注意：
# - kernel 用纯 CUDA 写（__global__ + blockIdx/threadIdx），cucc 自动转换
# - include <stdio.h>/<stdlib.h> 等照常写（shadow 遮蔽头处理）
# - 目前转换器 v0 只支持简单 __global__ void k(...)（无模板/函数指针参数）
# - 改了 src/ 或 bin/cucc 后需重新构建：bash tools/real_test/build_bridge.sh

# ---------- 5. 常见问题 ----------
# 5.1 cucc 报 "cannot execute: required file not found"
#     （CRLF 换行损坏，shebang 带 \r；run_tests 会自动修复，手动修如下）
sed -i 's/\r$//' build/real_test/install/bin/cucc

# 5.2 脚本被 Windows 传输转 CRLF（syntax error near ... $'do\r'）
sed -i 's/\r$//' tools/real_test/*.sh bin/cucc

# 5.3 改了 wrapper 源码（src/）后要重新构建
bash tools/real_test/build_bridge.sh

# 5.4 查看测试日志

cat build/real_test/logs/vectoradd_run.log
